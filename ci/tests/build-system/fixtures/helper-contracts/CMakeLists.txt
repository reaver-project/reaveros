cmake_minimum_required(VERSION 3.31)
project(helper-contracts LANGUAGES NONE)
include(ExternalProject)
include("${TEST_REPOSITORY}/cmake/target_functions.cmake")
set(REAVEROS_SOURCE_DIR "${TEST_REPOSITORY}")
set(REAVEROS_BINARY_DIR "${CMAKE_BINARY_DIR}")
add_custom_target(all-toolchain-prune)
add_custom_target(registered)
ExternalProject_Add(toolchain-example SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}"
    DOWNLOAD_COMMAND "" CONFIGURE_COMMAND "" BUILD_COMMAND "" INSTALL_COMMAND ""
    STEP_TARGETS install)

if (TEST_CASE STREQUAL "aggregate")
    reaveros_add_aggregate_targets(one two)
elseif (TEST_CASE STREQUAL "mode-aggregate")
    reaveros_add_aggregate_targets_with_modes(one two)
elseif (TEST_CASE STREQUAL "tags")
    reaveros_register_target(registered)
elseif (TEST_CASE STREQUAL "prune-source")
    reaveros_add_ep_prune_target(toolchain-example SOURCE_STEP)
elseif (TEST_CASE STREQUAL "prune-output")
    reaveros_add_ep_prune_target(toolchain-example REQUIRED_INSTALLED_OUTPUTS)
elseif (TEST_CASE STREQUAL "source")
    reaveros_add_ep_source_identity_step(toolchain-example fixture DEPENDEES)
elseif (TEST_CASE STREQUAL "tag")
    reaveros_add_ep_fetch_tag_target(toolchain-example revision extra)
else()
    message(FATAL_ERROR "Unknown helper contract case '${TEST_CASE}'.")
endif()
