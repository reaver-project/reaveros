cmake_minimum_required(VERSION 3.31)
project(child-cmake-contract CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

if (TEST_ROLE STREQUAL "provider")
    add_library(provided STATIC provided.cpp retained.cpp)
    target_compile_definitions(provided PRIVATE "PROVIDER_VALUE=${TEST_VALUE}")
    add_library(ordinary STATIC ordinary.cpp)
    configure_file(header.h.in header.h @ONLY)
    install(TARGETS provided ordinary DESTINATION .)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/header.h" DESTINATION .)
else()
    add_library(consumer SHARED consumer.cpp)
    target_include_directories(consumer PRIVATE "${TEST_PROVIDER}")
    target_link_directories(consumer PRIVATE "${TEST_PROVIDER}")
    target_link_options(consumer PRIVATE -nostdlib)
    target_link_libraries(consumer PRIVATE
        -Wl,--whole-archive provided -Wl,--no-whole-archive ordinary)
    reaveros_require_linker_dependencies(consumer CXX)
endif()
